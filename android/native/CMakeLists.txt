cmake_minimum_required(VERSION 3.22.1)
project(guanlan_talib C)
set(CMAKE_C_STANDARD 11)
file(GLOB TA_FUNCTIONS CONFIGURE_DEPENDS ta-lib/ta_func/*.c)
file(GLOB TA_COMMON CONFIGURE_DEPENDS ta-lib/ta_common/*.c)
file(GLOB TA_ABSTRACT CONFIGURE_DEPENDS ta-lib/ta_abstract/*.c ta-lib/ta_abstract/tables/*.c)
add_library(guanlan_talib SHARED ${TA_FUNCTIONS} ${TA_COMMON} ${TA_ABSTRACT} ta-lib/ta_abstract/frames/ta_frame.c)
target_include_directories(guanlan_talib PRIVATE ta-lib/include ta-lib/ta_common ta-lib/ta_func ta-lib/ta_abstract ta-lib/ta_abstract/frames)
target_compile_options(guanlan_talib PRIVATE -ffp-contract=off -fno-math-errno)
target_compile_definitions(guanlan_talib PRIVATE TA_LIB_SHARED)
target_link_libraries(guanlan_talib m)
target_link_options(guanlan_talib PRIVATE "-Wl,-z,max-page-size=16384")
